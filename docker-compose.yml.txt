version: "3.9"

services:
  api:
    build:
      context: ./apps/api
    container_name: seomind-api
    ports:
      - "8000:8000"
    volumes:
      - ./apps/api:/app
    command: uvicorn app.main:app --host 0.0.0.0 --port 8000 --reload
    depends_on:
      - db
      - redis

  db:
    image: postgres:16
    container_name: seomind-db
    restart: always
    environment:
      POSTGRES_DB: seomind
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: postgres
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data

  redis:
    image: redis:7
    container_name: seomind-redis
    ports:
      - "6379:6379"

  postgres_data:
